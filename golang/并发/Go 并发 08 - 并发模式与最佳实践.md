# Go 并发 08 — 并发模式与最佳实践

## Pipeline（流水线）

将处理拆分为多个阶段，通过 channel 连接：

```
gen(1,2,3) → sq → sq → print
```

```go
// 阶段1：生成数字
func gen(nums ...int) <-chan int {
    out := make(chan int)
    go func() {
        for _, n := range nums {
            out <- n
        }
        close(out)
    }()
    return out
}

// 阶段2：计算平方
func sq(in <-chan int) <-chan int {
    out := make(chan int)
    go func() {
        for n := range in {
            out <- n * n
        }
        close(out)
    }()
    return out
}

// 使用
for n := range sq(sq(gen(1, 2, 3))) {
    fmt.Println(n) // 1, 16, 81
}
```

## Fan-Out / Fan-In

### Fan-Out（扇出）

多个 goroutine 从同一个 channel 读取：

```go
func fanOut(in <-chan int, workers int) {
    var wg sync.WaitGroup
    for i := 0; i < workers; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            for n := range in {
                process(n)
            }
        }()
    }
    wg.Wait()
}
```

### Fan-In（扇入）

多个 channel 合并到一个 channel：

```go
func fanIn(chs ...<-chan int) <-chan int {
    out := make(chan int)
    var wg sync.WaitGroup

    // 为每个 input channel 启动一个 goroutine
    for _, ch := range chs {
        wg.Add(1)
        go func(c <-chan int) {
            defer wg.Done()
            for v := range c {
                out <- v
            }
        }(ch)
    }

    // 所有 goroutine 完成后关闭 out
    go func() {
        wg.Wait()
        close(out)
    }()

    return out
}
```

## 超时控制

```go
// 单个操作超时
func fetchWithTimeout(ctx context.Context, url string) (*Response, error) {
    ctx, cancel := context.WithTimeout(ctx, 5*time.Second)
    defer cancel()

    req, _ := http.NewRequestWithContext(ctx, "GET", url, nil)
    resp, err := http.DefaultClient.Do(req)
    if err != nil {
        return nil, err
    }
    return resp, nil
}
```

### 竞争超时（多个请求取最快）

```go
func fastest(urls []string) string {
    result := make(chan string)
    for _, url := range urls {
        go func(u string) {
            resp, _ := http.Get(u)
            if resp != nil {
                result <- u
            }
        }(url)
    }
    return <-result // 返回第一个完成的
}
```

## 优雅退出

```go
type Server struct {
    stopCh chan struct{}
}

func (s *Server) Start() {
    for {
        select {
        case job := <-s.jobCh:
            s.process(job)
        case <-s.stopCh:
            fmt.Println("shutting down...")
            return
        }
    }
}

func (s *Server) Stop() {
    close(s.stopCh) // 广播停止信号
}

// 捕获信号
func main() {
    srv := &Server{stopCh: make(chan struct{})}
    go srv.Start()

    sigCh := make(chan os.Signal, 1)
    signal.Notify(sigCh, syscall.SIGINT, syscall.SIGTERM)
    <-sigCh  // 等待 Ctrl+C

    srv.Stop()
    time.Sleep(1 * time.Second) // 等待清理
}
```

## 并发限流

### 方式1：channel 信号量

```go
sem := make(chan struct{}, 10) // 最大 10 并发

for _, task := range tasks {
    sem <- struct{}{}  // 获取令牌
    go func(t Task) {
        defer func() { <-sem }() // 释放
        process(t)
    }(task)
}

// 等待所有完成
for i := 0; i < cap(sem); i++ {
    sem <- struct{}{}
}
```

### 方式2：worker pool

```go
func workerPool(jobs <-chan Job, results chan<- Result, workers int) {
    var wg sync.WaitGroup
    for i := 0; i < workers; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            for job := range jobs {
                results <- process(job)
            }
        }()
    }
    wg.Wait()
    close(results)
}
```

## 数据一致性模式

```go
// 1. 共享变量 + Mutex
type Counter struct { mu sync.Mutex; v int }

// 2. channel + goroutine（CSP 推荐）
type Counter struct {
    ch    chan int
    value int
}

func NewCounter() *Counter {
    c := &Counter{ch: make(chan int)}
    go c.loop()
    return c
}

func (c *Counter) loop() {
    for op := range c.ch {
        c.value += op
    }
}

func (c *Counter) Inc() { c.ch <- 1 }
```
